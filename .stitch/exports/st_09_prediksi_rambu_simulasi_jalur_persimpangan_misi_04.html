Logika Percabangan If-Else 
# Aturan Penjelajahan FlipBot
sensor_rambu = True  # Status rambu batu kristal

if sensor_rambu == True:
    robot.belok_kanan("jembatan_emas")
    robot.ambil_hadiah_bintang()
else:
    robot.belok_kiri("gua_pelindung")
    robot.aktifkan_perisai()
