# kendali_cuaca.py — Terowongan Gelap Badai Kompilasi Sukses 
def kendali_cuaca_terowongan(sensor_hujan, sensor_kabut):
    # Syarat Mutlak Ganda (AND): Kedua sensor harus menyala
    if sensor_hujan and sensor_kabut:
        lampu = "BADAI_PENUH"
        kecepatan = 25
        return lampu, kecepatan  # [Lolos Uji ✓]

    # Syarat Toleran Alternatif (OR): Minimal salah satu menyala
    elif sensor_hujan or sensor_kabut:
        lampu = "SIAGA_KUNING"
        kecepatan = 40
        return lampu, kecepatan  # [Lolos Uji ✓]

    else:
        return "NORMAL_ECO", 60
